Experiment 9.1
Statement : Implement a Row-Level BEFORE UPDATE Trigger on the Salary_Hike table that restricts a salary
increase to no more than 15% of the :OLD.salary value; if the increase exceeds this limit, the trigger must raise a custom User-Defined Exception 
with a specific message

Code: 
-- 1. Create table
CREATE TABLE Salary_Hike (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(100),
    salary NUMERIC(10,2)
);

-- 2. Insert sample records
INSERT INTO Salary_Hike VALUES
(101, 'Amit', 30000),
(102, 'Rahul', 40000),
(103, 'Priya', 50000);

-- 3. Create trigger function
CREATE OR REPLACE FUNCTION CHECK_SALARY_HIKE()
RETURNS TRIGGER
AS $$
BEGIN
    IF NEW.salary > OLD.salary * 1.15 THEN
        RAISE EXCEPTION
        'Salary hike cannot exceed 15%% of the old salary';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 4. Create row-level BEFORE UPDATE trigger
CREATE OR REPLACE TRIGGER TRG_SALARY_HIKE
BEFORE UPDATE ON Salary_Hike
FOR EACH ROW
EXECUTE FUNCTION CHECK_SALARY_HIKE();

-- 5. Test the trigger
UPDATE Salary_Hike
SET salary = 34000
WHERE emp_id = 101;

-- 6. Display records
SELECT * FROM Salary_Hike;
